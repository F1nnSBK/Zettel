% --- RESEARCH EXPERIENCE & ENGINEERING PROJECTS ---
\section*{Research Experience \& Engineering Projects}

\cvitem{Kairos: Numerically Robust Contextual Bandits (Julia)}{2026}
\textit{Lead Researcher \& Developer | arXiv:2607.26832 | Accepted at SKILL 2026 (LNI)}
\vspace{-4pt}
\begin{itemize}[leftmargin=0.15in, label=\textbullet, itemsep=1.5pt, topsep=1.5pt]
  \item \textbf{Numerical Stability \& Cholesky Updates:} Developed \textbf{Kairos} in \textbf{Julia}, replacing standard Sherman-Morrison matrix inversion with direct rank-1 updates of Cholesky factors (\texttt{cholupdate}) in contextual Linear Upper Confidence Bound (LinUCB) bandits, preserving SPD matrix structure and preventing catastrophic condition number divergence under sparse item cold-start data ($\text{TTL} < 48\text{h}$).
  \item \textbf{Monte Carlo Stability Verification:} Validated numerical resilience across 1,000 Monte Carlo simulation runs, proving zero error variance in Cholesky updates compared to explosive divergence in traditional inversion.
  \item \textbf{Adaptive MRL Retrieval Subspace:} Integrated Matryoshka Representation Learning (MRL) for Maximum Inner Product Search (MIPS) candidate retrieval, reducing embedding dimensionality from 768d to 128d to achieve a \textbf{4.78$\times$ inference speedup} (0.065~ms vs 0.312~ms / 100 items) with an MAE of 0.032 (retaining $>96.8\%$ semantic structure).
  \item \textbf{Streaming Validation:} Deployed and benchmarked using live streaming data from the Tagesschau API.
\end{itemize}

\cvitem{Production-Grade Hybrid Recommendation Microservice (Project Work I)}{2025}
\textit{DHBW Ravensburg / Schw{\"a}bisch Media | Grade: 1.0 (Score: 97/100)}
\vspace{-4pt}
\begin{itemize}[leftmargin=0.15in, label=\textbullet, itemsep=1.5pt, topsep=1.5pt]
  \item \textbf{Cloud-Native Microservice Architecture:} Engineered an asynchronous FastAPI (ASGI) microservice on Google Cloud Platform orchestrating parallel prediction pipelines across Vertex AI Vector Search (3072d embeddings, MIPS) and Neural Collaborative Filtering (NCF / MLP).
  \item \textbf{Bayesian Hyperparameter Optimization:} Implemented a Tree-structured Parzen Estimator (Optuna TPE) search across 515,000 candidate evaluations (515 trials over 1,000 users), identifying an optimal 0.7226 / 0.2774 CBF-CF score-fusion equilibrium.
  \item \textbf{Rigorously Unbiased Evaluation:} Evaluated performance under a Leave-Last-Out Full-Catalog Ranking protocol over 2.3M users and 104k items ($<0.005\%$ matrix density), proving a \textbf{5.4$\times$ nDCG@10 improvement} over baseline models at a 95th-percentile latency of 1,562~ms (SLO $< 2,000$~ms).
\end{itemize}
